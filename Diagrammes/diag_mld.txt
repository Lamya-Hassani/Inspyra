@startuml
!theme plain
skinparam linetype ortho
skinparam class {
    BackgroundColor White
    ArrowColor #2E7D32
    BorderColor #2E7D32
}

    class Utilisateur {
        + id : int [PK]
        --
        username : varchar(150)
        email : varchar(254)
        password : varchar(128)
        role : varchar(20) {SUPERADMIN, ADMIN, CLIENT}
        telephone : varchar(20)
        adresse : varchar(255)
        ville : varchar(100)
        codePostal : varchar(20)
        pays : varchar(100)
        dateCreation : datetime
        is_staff : boolean
        is_superuser : boolean
    }

    class UserPreference {
        + id : int [PK]
        --
        # user_id : int [FK]
        light_level : varchar(10)
        watering_frequency : varchar(10)
        has_pets : boolean
        has_children : boolean
        experience_level : varchar(15)
        primary_goal : varchar(20)
    }
    class Categorie {
        + id : int [PK]
        --
        nom : varchar(100)
    }

    class Plante {
        + id : int [PK]
        --
        # categorie_id : int [FK]
        nom : varchar(100)
        nomScientifique : varchar(200)
        prix : decimal(10,2)
        stock : int
        description : text
        besoinEau : varchar(100)
        besoinLumiere : varchar(100)
        temperatureMin : float
        temperatureMax : float
        humidite : float
        typeSol : varchar(100)
        niveauEntretien : varchar(100)
        image : varchar(100)
    }
    class Commande {
        + id : int [PK]
        --
        # utilisateur_id : int [FK]
        date : datetime
        statut : varchar(20) {PENDING, SHIPPED, DELIVERED, CANCELLED}
        total : decimal(10,2)
    }

    class LigneCommande {
        + id : int [PK]
        --
        # commande_id : int [FK]
        # plante_id : int [FK]
        quantite : int
        prix : decimal(10,2)
    }

    class Paiement {
        + id : int [PK]
        --
        # commande_id : int [FK]
        montant : decimal(10,2)
        methode : varchar(20) {CARD, PAYPAL, COD}
        statut : varchar(20) {PENDING, COMPLETED, FAILED, REFUNDED}
        transaction_id : varchar(100)
        date : datetime
    }

    class Panier {
        + id : int [PK]
        --
        # utilisateur_id : int [FK]
        date_creation : datetime
        date_modification : datetime
    }

    class ArticlePanier {
        + id : int [PK]
        --
        # panier_id : int [FK]
        # plante_id : int [FK]
        quantite : int
    }

' Relations
Utilisateur "1" -- "0..1" UserPreference : possède
Utilisateur "1" -- "0..*" Commande : passe
Utilisateur "1" -- "1" Panier : possède

Categorie "1" -- "0..*" Plante : contient

Commande "1" -- "1..*" LigneCommande : contient
Commande "1" -- "1" Paiement : réglée par

Plante "1" -- "0..*" LigneCommande : incluse dans
Plante "1" -- "0..*" ArticlePanier : ajoutée dans

Panier "1" -- "0..*" ArticlePanier : contient

@enduml
