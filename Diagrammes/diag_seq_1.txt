@startuml
title Séquence — Connexion (regroupement backend)
actor "Personne" as U
participant "Frontend" as F
participant "Backend" as Backend
database "Base de donnée" as DB
U -> F : identifiants
F -> Backend : login(username, password)
alt utilisateur valide
  Backend -> Backend : traiter la demande
  Backend -> Backend : préparer les jetons
  Backend -> DB : lire Utilisateur
  DB --> Backend : utilisateur trouvé
  Backend --> Backend : jetons + informations rôle
  Backend --> F : accès autorisé (jetons)
  F --> U : accès autorisé
else utilisateur non trouvé ou mot de passe incorrect
  Backend -> Backend : traiter la demande
  Backend -> DB : lire Utilisateur
  DB --> Backend : aucun utilisateur / échec authentification
  Backend --> F : refus (erreur 401 ou 400)
  F --> U : accès refusé
end
